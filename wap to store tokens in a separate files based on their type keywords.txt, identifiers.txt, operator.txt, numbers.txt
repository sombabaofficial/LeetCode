f = open("input.txt", "r")
data = f.read()

keywords = ['if', 'else', 'while', 'int', 'float', 'return']
operators = ['+', '-', '*', '/', '=', '==', '!=', '<', '>', '<=', '>=']

words = data.split()

k = open("keywords.txt", "w")
i = open("identifiers.txt", "w")
o = open("operators.txt", "w")
n = open("numbers.txt", "w")

for w in words:
    if w in keywords:
        k.write(w + "\n")
    elif w in operators:
        o.write(w + "\n")
    elif w.isdigit():
        n.write(w + "\n")
    elif w.isalnum():
        i.write(w + "\n")

k.close()
i.close()
o.close()
n.close()

print("Tokens stored in files")
